import java.io.*;
import java.util.*;

/**
 * Markowitz Portfolio Optimization Engine
 * Calculates returns, standard deviations, and Variance-Covariance matrix
 * for asset allocation and efficient frontier optimization.
 */
public class Calculate {

    // 1. Arvutab instrumentide päeva- ja aastased tulumäärad ning volatiilsuse
    static Map<String, Map<String, Double>> arvutaStatistika(Map<String, List<Double>> ratesMap) {
        Map<String, Map<String, Double>> statsMap = new LinkedHashMap<>();

        for (String instrument : ratesMap.keySet()) {
            List<Double> dailyReturns = ratesMap.get(instrument);

            // Päevase keskmise tulumäära arvutamine
            double avgDailyReturn = dailyReturns.stream().mapToDouble(Double::doubleValue).average().orElse(0.0);

            // Päevase standardhälbe (volatiilsuse) arvutamine
            double dailyStdDev = Math.sqrt(dailyReturns.stream()
                    .mapToDouble(r -> Math.pow(r - avgDailyReturn, 2))
                    .average().orElse(0.0));

            // Aastane tulumäär (252 kauplemispäeva)
            double annualizedReturn = avgDailyReturn * 252;
            double annualizedStdDev = dailyStdDev * Math.sqrt(252);

            Map<String, Double> instrumentStats = new LinkedHashMap<>();
            instrumentStats.put("Daily Return Rate", avgDailyReturn);
            instrumentStats.put("Annualized Return Rate", annualizedReturn);
            instrumentStats.put("Daily Standard Deviation", dailyStdDev);
            instrumentStats.put("Annualized Standard Deviation", annualizedStdDev);

            statsMap.put(instrument, instrumentStats);
        }
        return statsMap;
    }

    // 2. Arvutab instrumentide vahelise VAR-COVAR (kovariatsiooni) maatriksi
    public static double[][] arvutaVarCovarMaatriks(Map<String, List<Double>> rates, 
                                                    Map<String, Map<String, Double>> stats) {
        List<String> instruments = new ArrayList<>(rates.keySet());
        int n = rates.get(instruments.get(0)).size();
        double[][] meanCentered = new double[instruments.size()][n];

        // Normaliseerime andmed: lahutame igast väärtusest keskmise
        for (int i = 0; i < instruments.size(); i++) {
            String instrument = instruments.get(i);
            List<Double> returns = rates.get(instrument);
            double mean = stats.get(instrument).get("Daily Return Rate");
            for (int j = 0; j < n; j++) {
                meanCentered[i][j] = returns.get(j) - mean;
            }
        }

        // VAR-COVAR maatriksi arvutamine ja aastaseks taandamine (divisor = n / 252.0)
        double[][] varCovar = new double[instruments.size()][instruments.size()];
        double divisor = n / 252.0;
        for (int i = 0; i < instruments.size(); i++) {
            for (int j = 0; j < instruments.size(); j++) {
                double sum = 0.0;
                for (int k = 0; k < n; k++) {
                    sum += meanCentered[i][k] * meanCentered[j][k];
                }
                varCovar[i][j] = sum / divisor;
            }
        }
        return varCovar;
    }

    // 3. Optimeerib portfelli ja genereerib tulemused CSV formaati
    public static void teePortfelliOptimeerimine(Map<String, Map<String, Double>> stats, 
                                                 double[][] varCovarMaatriks) {
        List<String> assets = new ArrayList<>(stats.keySet());
        String fileName = "Optimeerimine.csv";
        
        try (PrintWriter writer = new PrintWriter(new FileWriter(fileName))) {
            writer.print("Asset");
            for (String asset : assets) writer.print("," + asset);
            writer.println();
            
            writer.print("Annualized Return Rate");
            for (String asset : stats.keySet())
                writer.print("," + stats.get(asset).get("Annualized Return Rate"));
            writer.println();

            // Portfelli kaalud ja Excel Solver optimeerimise valemid
            writer.print("Weight");
            for (int i = 0; i < assets.size(); i++) writer.print("," + (i == 0 ? "1" : "0"));
            writer.println();
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
