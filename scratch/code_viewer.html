def loe_uus_therand_reservations():
    """Loeb The Randi reaalajas API kaudu reserveeringuandmed."""
    try:
        params = {"key": NEW_API_KEY}
        response = requests.get(NEW_API_URL, params=params, timeout=10)
        response.raise_for_status()
        return response.json()
    except Exception as e:
        print(f"[VIGA] API päring ebaõnnestus: {e}")
        return []

def lylita_valjak(valjaku_nr, olek):
    """Juhib Tridonic valgustuskontrollerit (DALI käsud)."""
    valjaku_id = VALJAKUTE_ID_TABEL.get(valjaku_nr)
    if not valjaku_id:
        return
    
    url = f"http://{CONTROLLER_IP}/api/circuits/{valjaku_id}/state"
    headers = {"Authorization": f"Bearer {TRIDONIC_AUTH_KEY}"}
    payload = {"on": olek, "fade_time": 2.0}
    
    try:
        res = requests.post(url, json=payload, headers=headers, timeout=5)
        print(f"[VALGUSTUS] Väljak {valjaku_nr} -> {'SEES' if olek else 'VÄLJAS'} ({res.status_code})")
    except Exception as err:
        print(f"[TÕRGE] Kontrolleriga ühendus katkes: {err}")
